#   apt install -y python3.12-venv nvidia-cuda-toolkit
#   python3 -m venv ~/symbolicai-env
#   source ~/symbolicai-env/bin/activate
#   pip install --upgrade pip setuptools wheel
#
#   # 1. PyTorch must come from the pytorch cu121 index (default PyPI is CPU-only)
#   pip install torch==2.4.0 --index-url https://download.pytorch.org/whl/cu121
#
#   # 2. causal-conv1d and mamba-ssm compile CUDA C++ extensions from source.
#   #    - Must use --no-build-isolation so they link against our torch 2.4.0,
#   #      not whatever pip's isolated build env pulls in.
#   #    - TORCH_CUDA_ARCH_LIST restricts compilation to the target architecture.
#   #      Without it, mamba-ssm builds 8 architectures (sm_53..sm_90) and OOMs
#   #      on a 7.6 GB WSL2. Set it to match your GPU (see table below).
#   #    - MAX_JOBS=2 keeps peak RAM under control.
#   export TORCH_CUDA_ARCH_LIST="8.9"   # RTX 4070/4080/4090 (Ada Lovelace)
#   export MAX_JOBS=2
#   export CUDA_HOME=/usr
#   pip install --no-build-isolation causal-conv1d
#   pip install --no-build-isolation mamba-ssm
#
# GPU → TORCH_CUDA_ARCH_LIST cheat sheet:
#   RTX 30xx (Ampere)          -> "8.6"
#   RTX 40xx (Ada Lovelace)    -> "8.9"
#   H100 (Hopper)              -> "9.0"
#   A100 (Ampere datacenter)   -> "8.0"
#
